[	A - Nine or Nein](https://atcoder.jp/contests/abc471/tasks/abc471_a)
```
function Main(input) {
  const [a, b] = input.trim().split(" ");
  const A = Number(a);
  const B = Number(b);

  if ((A + B) === 9) {
    console.log("Nine");
  } else if ((A - B) === 9) {
    console.log("Nine");
  } else if ((A * B) === 9) {
    console.log("Nine");
  } else if ((A / B) === 9) {
    console.log("Nine");
  } else {
    console.log("Nein");
  }
}

Main(require("fs").readFileSync("/dev/stdin", "utf8"));

[B - Survey Tabulation](https://atcoder.jp/contests/abc471/tasks/abc471_b)
```
function Main(input) {
  const arr = input.trim().split("\n");
  const a = arr.shift();
  const b = arr;
  const unique = [...new Set(b.map(s => s.toLowerCase()))]
  let array = [];
  for (let i = 0; i <= unique.length - 1; i++) {
    let count = 0;
    for (let g = 0; g <= b.length - 1; g++) {
        const crr = b[g]
        if (unique[i] === crr.toLowerCase()) {
            count = count + 1;
        }
    }
    array[i] = [unique[i], count]
  }
  array.sort((a, b) => b[1] - a[1])
  console.log(array[0][1]);
}
